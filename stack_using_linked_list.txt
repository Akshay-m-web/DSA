// stack using Linked list
#include <stdio.h>
#include <stdlib.h>
struct node{
    int data;
    struct node*next;
}*top=NULL;
void push(int x){
    struct node*t=(struct node*)malloc(sizeof(struct node));
    if(t==NULL){
        printf("stack overflow\n");
    }
    else{
        t->data=x;
        t->next=top;
        top=t;
    }
}
int pop(){
    int x=-1;
    if(top==NULL){
        printf("stack is empty\n");
    }
    else{
        x=top->data;
        struct node*t=top;
        top=top->next;
        free(t);
    }
    return x;
}
void display(){
    if(top==NULL){
        printf("stack is empty\n");
    }
    else{
       struct node*t=top;
       while(t){
           printf("%d ",t->data);
           t=t->next;
       }
       printf("\n");
    }
}
int peek(int pos){
    int x=-1;
     if(top==NULL){
        printf("stack is empty\n");
    }
    else{
       struct node*t=top;
        int i;
        for(i=0;i<(pos-1)&&t;i++){
            t=t->next;
        }
        if(t){
        x=t->data;
        }
        return x;
        
    }
       
}
int main() {
   push(10);
    push(20);
     push(30);
      push(40);
       push(50);
      display();
      printf("%d\n",pop());
      display();
      printf("%d ",peek(3));
      
    return 0;
}
